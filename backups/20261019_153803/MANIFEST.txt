Restore:
  # 1. bring up postgres
  docker compose -p idm -f docker-compose.procurement.prod.yml up -d postgres

  # 2. restore the dump
  cat database.dump | docker compose -p idm -f docker-compose.procurement.prod.yml exec -T postgres \
    pg_restore -U "$POSTGRES_USER" -d "$POSTGRES_DB" --clean --if-exists

  # 3. (optional) restore media files to the volume
  docker run --rm -v idm_media_files:/dest -v .:/backup alpine \
    tar -xzf /backup/media_files.tar.gz -C /dest

  # 4. restart the full stack
  docker compose -p idm -f docker-compose.procurement.prod.yml up -d
